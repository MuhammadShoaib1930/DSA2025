   string solved(string word) {
    string w1 = "";
    string w2 = "";
    bool isFirst = true;
    bool isPluse = false;
    for (auto&& i : word)
    {
        if (i == '@') {
            isFirst = false;
        }
        if (isFirst) {
            if (i == '+' || isPluse) {
                isPluse = true;
            }
            else {
                if (i != '.') {
                    w1 += i;
                }

            }

        }
        else {
            w2 += i;
        }
    }
    return (w1 + w2);
}
int numUniqueEmails(vector<string>& emails) {
    // .com
    unordered_set<string> res;
    for (string& email : emails)
    {
        string w = solved(email);
        res.insert(w);

    }
    return res.size();
}
